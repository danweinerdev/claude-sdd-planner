func TestGraphSyncCostJSONPreservesRecordedResultOnReleaseError(t *testing.T) {
	cost := evidencecost.New(nil)
	res := &gsync.Result{Node: "work", Recorded: true, Merged: true, AttemptID: "at-00000000000000000000000000000000", Observation: &model.Verification{Result: model.ResultPass, Seq: 7}}
	path := filepath.Join(t.TempDir(), "stdout.json")
	f, err := os.OpenFile(path, os.O_CREATE|os.O_WRONLY|os.O_TRUNC, 0o600)
	if err != nil {
		t.Fatal(err)
	}
	oldStdout := os.Stdout
	os.Stdout = f
	defer func() { os.Stdout = oldStdout }()
	releaseErr := errors.New("release failed")
	gotErr := emitGraphSyncCostError(nil, true, res, releaseErr, cost)
	if closeErr := f.Close(); closeErr != nil {
		t.Fatal(closeErr)
	}
	if !errors.Is(gotErr, releaseErr) {
		t.Fatalf("error = %v, want release failure", gotErr)
	}
	raw, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	var got struct {
		Recorded    bool                  `json:"recorded"`
		Merged      bool                  `json:"merged"`
		AttemptID   string                `json:"attempt_id"`
		Observation *model.Verification   `json:"observation"`
		Cost        *evidencecost.Summary `json:"cost"`
	}
	if err := json.Unmarshal(raw, &got); err != nil {
		t.Fatalf("decode result: %v\n%s", err, raw)
	}
	if !got.Recorded || !got.Merged || got.AttemptID != res.AttemptID || got.Observation == nil || got.Observation.Result != model.ResultPass || got.Observation.Seq != 7 || got.Cost == nil {
		t.Fatalf("partial recorded result was discarded: %+v", got)
	}
}

func TestGraphSyncCostJSONDoesNotHideWriterFailure(t *testing.T) {
	f, err := os.CreateTemp(t.TempDir(), "closed-stdout")
	if err != nil {
		t.Fatal(err)
	}
	if err := f.Close(); err != nil {
		t.Fatal(err)
	}
	oldStdout := os.Stdout
	os.Stdout = f
	defer func() { os.Stdout = oldStdout }()
	releaseErr := errors.New("release failed")
	gotErr := emitGraphSyncCostError(nil, true, &gsync.Result{Recorded: true}, releaseErr, evidencecost.New(nil))
	if !errors.Is(gotErr, releaseErr) || !strings.Contains(gotErr.Error(), "writing graph sync cost JSON") {
		t.Fatalf("writer failure not joined with operation error: %v", gotErr)
	}
}

func TestCostIntegrationRealGoCaptureRefusalsAndReplay(t *testing.T) {
	root := cliEvidenceFixture(t, false)
	old, _ := os.Getwd()
	if err := os.Chdir(root); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = os.Chdir(old) })

	err, out := executeEvidenceCLI(t, "test", "run", "--plan", "Pilot", "--node", "work", "--by", "worker", "--phase", "green", "--cost")
	if exitCode(err) != 0 || !strings.Contains(out, "cost: total_ns=") || !strings.Contains(out, "test_execution_requests=1") {
		t.Fatalf("cost run exit=%d err=%v out=%s", exitCode(err), err, out)
	}
	attempt := strings.TrimSuffix(strings.Fields(out)[1], ":")
	header := filepath.Join(root, "Plans", "Pilot", ".graph", "test-evidence", "work", attempt, "header.json")
	if raw := readEvidenceCLI(t, header); strings.Contains(raw, `"cost"`) {
		t.Fatal("cost persisted in raw attempt header")
	}

	err, out = executeEvidenceCLI(t, "test", "check", "--plan", "Pilot", "--node", "work", "--attempt", attempt, "--expect", "green")
	if exitCode(err) != 0 || strings.Contains(out, "cost:") {
		t.Fatalf("default check changed: err=%v out=%s", err, out)
	}

	stdoutPath := filepath.Join(filepath.Dir(header), "stdout.json")
	rawOutput, err := os.ReadFile(stdoutPath)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(stdoutPath, append(rawOutput, '\n'), 0o600); err != nil {
		t.Fatal(err)
	}
	err, out = executeEvidenceCLI(t, "test", "check", "--plan", "Pilot", "--node", "work", "--attempt", attempt, "--expect", "green", "--cost")
	if exitCode(err) != 1 || !strings.Contains(out, "cost:") {
		t.Fatalf("tamper refusal changed: exit=%d err=%v out=%s", exitCode(err), err, out)
	}
	if err := os.WriteFile(stdoutPath, rawOutput, 0o600); err != nil {
		t.Fatal(err)
	}

	source := filepath.Join(root, "subject_test.go")
	rawSource, err := os.ReadFile(source)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(source, append(rawSource, []byte("// changed\n")...), 0o644); err != nil {
		t.Fatal(err)
	}
	err, _ = executeEvidenceCLI(t, "test", "check", "--plan", "Pilot", "--node", "work", "--attempt", attempt, "--expect", "green", "--cost")
	if exitCode(err) != 1 {
		t.Fatalf("changed-source refusal exit=%d err=%v", exitCode(err), err)
	}
	if err := os.WriteFile(source, rawSource, 0o644); err != nil {
		t.Fatal(err)
	}

	graphPath := gstore.PathFor(filepath.Join(root, "Plans", "Pilot"))
	if _, err := gstore.Update(graphPath, func(g *model.Graph) error { g.NodeByID("work").Claim.Instance = "changed"; return nil }); err != nil {
		t.Fatal(err)
	}
	err, _ = executeEvidenceCLI(t, "test", "check", "--plan", "Pilot", "--node", "work", "--attempt", attempt, "--expect", "green", "--cost")
	if exitCode(err) != 1 {
		t.Fatalf("changed-claim refusal exit=%d err=%v", exitCode(err), err)
	}
	if _, err := gstore.Update(graphPath, func(g *model.Graph) error { g.NodeByID("work").Claim.Instance = "one"; return nil }); err != nil {
		t.Fatal(err)
	}

	err, out = executeEvidenceCLI(t, "graph", "sync", "--plan", "Pilot", "--node", "work", "--by", "worker", "--attempt", attempt, "--cost")
	if exitCode(err) != 0 || !strings.Contains(out, "graph_load_requests=") {
		t.Fatalf("cost sync exit=%d err=%v out=%s", exitCode(err), err, out)
	}
	if raw := readEvidenceCLI(t, graphPath); strings.Contains(raw, `"cost"`) {
		t.Fatal("cost persisted in graph")
	}
	err, out = executeEvidenceCLI(t, "graph", "sync", "--plan", "Pilot", "--node", "work", "--attempt", attempt, "--cost")
	if exitCode(err) != 0 || !strings.Contains(out, "historical_replays=1") {
		t.Fatalf("cost replay exit=%d err=%v out=%s", exitCode(err), err, out)
	}
}
