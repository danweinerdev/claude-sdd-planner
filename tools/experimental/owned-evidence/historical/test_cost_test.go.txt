func TestTestCostIncompleteCaptureRetainsAttemptID(t *testing.T) {
	root := cliEvidenceFixture(t, false)
	old, err := os.Getwd()
	if err != nil {
		t.Fatal(err)
	}
	if err := os.Chdir(root); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = os.Chdir(old) })
	if err := os.WriteFile(filepath.Join(root, "subject_test.go"), []byte("package pilot\nimport (\"testing\";\"time\")\nfunc TestValue(t *testing.T){time.Sleep(time.Hour)}\n"), 0o644); err != nil {
		t.Fatal(err)
	}
	planDir := filepath.Join(root, "Plans", "Pilot")
	if _, err := gstore.Update(gstore.PathFor(planDir), func(g *model.Graph) error {
		g.NodeByID("work").Gate.Execution.TimeoutSeconds = 1
		return nil
	}); err != nil {
		t.Fatal(err)
	}
	err, out := executeEvidenceCLI(t, "test", "run", "--plan", "Pilot", "--node", "work", "--by", "worker", "--phase", "green", "--cost")
	if exitCode(err) != 2 || !strings.HasPrefix(out, "attempt at-") || !strings.Contains(out, "execution-incomplete") || !strings.Contains(out, "cost:") {
		t.Fatalf("incomplete capture lost its status or cost: exit=%d err=%v out=%s", exitCode(err), err, out)
	}
	attempt := strings.TrimSuffix(strings.Fields(out)[1], ":")
	raw, err := os.ReadFile(filepath.Join(planDir, ".graph", "test-evidence", "work", attempt, "incomplete.json"))
	if err != nil {
		t.Fatal(err)
	}
	if strings.Contains(string(raw), `"cost"`) {
		t.Fatal("diagnostic cost persisted into incomplete attempt")
	}
}

func TestTestCostFlagIsOptionalOnRunAndCheck(t *testing.T) {
	for _, path := range []string{"sdd test run", "sdd test check"} {
		cmd := findByPath(newRootCmd(), path)
		if cmd == nil {
			t.Fatalf("command %q is missing", path)
		}
		if flag := cmd.Flags().Lookup("cost"); flag == nil {
			t.Errorf("%s has no --cost flag", path)
		} else if flag.DefValue != "false" {
			t.Errorf("%s --cost default = %q, want false", path, flag.DefValue)
		}
	}
}

func TestTestCostOutputContract(t *testing.T) {
	root := cliEvidenceFixture(t, false)
	old, _ := os.Getwd()
	if err := os.Chdir(root); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = os.Chdir(old) })

	runJSON, err := testCostJSON(t, "test", "run", "--plan", "Pilot", "--node", "work", "--by", "worker", "--phase", "green", "--cost", "--json")
	if exitCode(err) != 0 {
		t.Fatalf("cost run exit=%d err=%v output=%s", exitCode(err), err, runJSON)
	}
	var runEnvelope map[string]json.RawMessage
	if err := json.Unmarshal([]byte(runJSON), &runEnvelope); err != nil {
		t.Fatalf("decode cost run JSON: %v\n%s", err, runJSON)
	}
	rawRunCost, ok := runEnvelope["cost"]
	if !ok {
		t.Fatalf("cost run JSON has no cost object: %s", runJSON)
	}
	assertTestCostNumericOnly(t, rawRunCost, root, "worker", "subject_test.go")
	var run struct {
		AttemptID       string                `json:"attempt_id"`
		ExecutionStatus string                `json:"execution_status"`
		Result          string                `json:"result"`
		Complete        bool                  `json:"complete"`
		Cost            *evidencecost.Summary `json:"cost"`
	}
	if err := json.Unmarshal([]byte(runJSON), &run); err != nil {
		t.Fatal(err)
	}
	if run.AttemptID == "" || run.ExecutionStatus != "passed" || run.Result != "pass" || !run.Complete || run.Cost == nil {
		t.Fatalf("unexpected run result: %+v", run)
	}
	assertTestCostNonnegative(t, *run.Cost)
	if _, ok := run.Cost.PhaseNS[evidencecost.RootInputPreparation]; !ok {
		t.Fatalf("run cost excludes root/input preparation: %+v", run.Cost.PhaseNS)
	}
	if _, ok := run.Cost.PhaseNS[evidencecost.TestExecution]; !ok {
		t.Fatalf("run cost lacks test_execution timing: %+v", run.Cost.PhaseNS)
	}
	if run.Cost.Counters[evidencecost.TestExecutionRequests] != 1 || run.Cost.Counters[evidencecost.TestProcessesCompleted] != 1 {
		t.Fatalf("run did not attribute its one completed test process: %+v", run.Cost.Counters)
	}
	for _, counter := range []evidencecost.Counter{evidencecost.ProfileResolutions, evidencecost.GoProbeRequests, evidencecost.RootHashRequests, evidencecost.ReportParses} {
		if run.Cost.Counters[counter] == 0 {
			t.Fatalf("run cost lacks %s: %+v", counter, run.Cost.Counters)
		}
	}

	header := filepath.Join(root, "Plans", "Pilot", ".graph", "test-evidence", "work", run.AttemptID, "header.json")
	if raw := readEvidenceCLI(t, header); strings.Contains(raw, `"cost"`) {
		t.Fatal("cost persisted in immutable receipt")
	}

	defaultJSON, err := testCostJSON(t, "test", "check", "--plan", "Pilot", "--node", "work", "--attempt", run.AttemptID, "--expect", "green", "--json")
	if exitCode(err) != 0 {
		t.Fatalf("default check exit=%d err=%v output=%s", exitCode(err), err, defaultJSON)
	}
	var defaultEnvelope map[string]json.RawMessage
	if err := json.Unmarshal([]byte(defaultJSON), &defaultEnvelope); err != nil {
		t.Fatal(err)
	}
	if _, ok := defaultEnvelope["cost"]; ok {
		t.Fatalf("check without --cost emitted cost: %s", defaultJSON)
	}

	checkJSON, err := testCostJSON(t, "test", "check", "--plan", "Pilot", "--node", "work", "--attempt", run.AttemptID, "--expect", "green", "--cost", "--json")
	if exitCode(err) != 0 {
		t.Fatalf("cost check exit=%d err=%v output=%s", exitCode(err), err, checkJSON)
	}
	var checkEnvelope map[string]json.RawMessage
	if err := json.Unmarshal([]byte(checkJSON), &checkEnvelope); err != nil {
		t.Fatal(err)
	}
	rawCheckCost, ok := checkEnvelope["cost"]
	if !ok {
		t.Fatalf("cost check JSON has no cost object: %s", checkJSON)
	}
	assertTestCostNumericOnly(t, rawCheckCost, root, run.AttemptID, "subject_test.go")
	var check struct {
		Eligible bool                  `json:"eligible"`
		Cost     *evidencecost.Summary `json:"cost"`
	}
	if err := json.Unmarshal([]byte(checkJSON), &check); err != nil {
		t.Fatal(err)
	}
	if !check.Eligible || check.Cost == nil {
		t.Fatalf("unexpected check result: %+v", check)
	}
	assertTestCostNonnegative(t, *check.Cost)
	if check.Cost.Counters[evidencecost.TestExecutionRequests] != 0 || check.Cost.Counters[evidencecost.TestProcessesCompleted] != 0 {
		t.Fatalf("read-only check attributed test execution: %+v", check.Cost.Counters)
	}
	for _, counter := range []evidencecost.Counter{evidencecost.GraphLoadRequests, evidencecost.BundleReadRequests, evidencecost.ProfileResolutions, evidencecost.ReportParses} {
		if check.Cost.Counters[counter] == 0 {
			t.Fatalf("check cost lacks %s: %+v", counter, check.Cost.Counters)
		}
	}

	errorJSON, err := testCostJSON(t, "test", "check", "--plan", "Missing", "--node", "work", "--attempt", run.AttemptID, "--expect", "green", "--cost", "--json")
	if exitCode(err) == 0 {
		t.Fatalf("missing-plan check unexpectedly succeeded: %s", errorJSON)
	}
	var failed struct {
		Error string                `json:"error"`
		Cost  *evidencecost.Summary `json:"cost"`
	}
	if err := json.Unmarshal([]byte(errorJSON), &failed); err != nil {
		t.Fatalf("decode started-operation error JSON: %v\n%s", err, errorJSON)
	}
	if failed.Error == "" || failed.Cost == nil {
		t.Fatalf("started-operation error lacks error or cost: %+v", failed)
	}
	if _, ok := failed.Cost.PhaseNS[evidencecost.RootInputPreparation]; !ok {
		t.Fatalf("resolution error excludes preparation phase: %+v", failed.Cost.PhaseNS)
	}
}

func testCostJSON(t *testing.T, args ...string) (string, error) {
	t.Helper()
	path := filepath.Join(t.TempDir(), "stdout.json")
	f, openErr := os.OpenFile(path, os.O_CREATE|os.O_WRONLY|os.O_TRUNC, 0o600)
	if openErr != nil {
		t.Fatal(openErr)
	}
	oldStdout := os.Stdout
	os.Stdout = f
	defer func() { os.Stdout = oldStdout }()
	var commandOutput string
	err, out := executeEvidenceCLI(t, args...)
	commandOutput = out
	if closeErr := f.Close(); closeErr != nil {
		t.Fatal(closeErr)
	}
	os.Stdout = oldStdout
	stdout, readErr := os.ReadFile(path)
	if readErr != nil {
		t.Fatal(readErr)
	}
	return string(stdout) + commandOutput, err
}

func assertTestCostNonnegative(t *testing.T, cost evidencecost.Summary) {
	t.Helper()
	if cost.TotalNS < 0 {
		t.Fatalf("negative total_ns: %d", cost.TotalNS)
	}
	for phase, nanos := range cost.PhaseNS {
		if nanos < 0 {
			t.Fatalf("negative phase %s: %d", phase, nanos)
		}
	}
}

func assertTestCostNumericOnly(t *testing.T, raw json.RawMessage, forbidden ...string) {
	t.Helper()
	var value any
	if err := json.Unmarshal(raw, &value); err != nil {
		t.Fatal(err)
	}
	object, ok := value.(map[string]any)
	if !ok {
		t.Fatalf("cost is %T, want object", value)
	}
	for _, key := range []string{"total_ns", "phase_ns", "counters"} {
		if _, ok := object[key]; !ok {
			t.Errorf("cost object lacks %q: %s", key, raw)
		}
	}
	if len(object) != 3 {
		t.Errorf("cost object keys = %v, want only total_ns, phase_ns, counters", object)
	}
	var walk func(any)
	walk = func(v any) {
		switch typed := v.(type) {
		case map[string]any:
			for _, child := range typed {
				walk(child)
			}
		case float64:
		case string:
			t.Errorf("cost contains string value %q", typed)
		default:
			t.Errorf("cost contains nonnumeric value of type %T", typed)
		}
	}
	walk(value)
	text := string(raw)
	for _, secret := range forbidden {
		if secret != "" && strings.Contains(text, secret) {
			t.Errorf("cost contains invocation detail %q: %s", secret, text)
		}
	}
}
