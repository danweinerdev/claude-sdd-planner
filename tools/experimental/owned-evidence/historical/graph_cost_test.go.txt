func TestGraphSyncCostFlagIsOptional(t *testing.T) {
	cmd := findByPath(newRootCmd(), "sdd graph sync")
	if cmd == nil {
		t.Fatal("sdd graph sync is missing")
	}
	if flag := cmd.Flags().Lookup("cost"); flag == nil {
		t.Fatal("sdd graph sync has no --cost flag")
	} else if flag.DefValue != "false" {
		t.Fatalf("sdd graph sync --cost default = %q, want false", flag.DefValue)
	}
}

func TestGraphSyncCostOutputContract(t *testing.T) {
	root := cliEvidenceFixture(t, false)
	old, _ := os.Getwd()
	if err := os.Chdir(root); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = os.Chdir(old) })

	attempt := graphCostCaptureAttempt(t)
	planDir := filepath.Join(root, "Plans", "Pilot")
	graphPath := gstore.PathFor(planDir)
	syncJSON, err := graphCostJSON(t, "graph", "sync", "--plan", "Pilot", "--node", "work", "--by", "worker", "--attempt", attempt, "--cost", "--json")
	if exitCode(err) != 0 {
		t.Fatalf("cost sync exit=%d err=%v output=%s", exitCode(err), err, syncJSON)
	}
	var syncEnvelope map[string]json.RawMessage
	if err := json.Unmarshal([]byte(syncJSON), &syncEnvelope); err != nil {
		t.Fatalf("decode cost sync JSON: %v\n%s", err, syncJSON)
	}
	rawSyncCost, ok := syncEnvelope["cost"]
	if !ok {
		t.Fatalf("cost sync JSON has no cost object: %s", syncJSON)
	}
	assertGraphCostNumericOnly(t, rawSyncCost, root, attempt, "worker", "subject_test.go")
	var synced struct {
		Recorded    bool                  `json:"recorded"`
		Historical  bool                  `json:"historical"`
		AttemptID   string                `json:"attempt_id"`
		Observation *model.Verification   `json:"observation"`
		Cost        *evidencecost.Summary `json:"cost"`
	}
	if err := json.Unmarshal([]byte(syncJSON), &synced); err != nil {
		t.Fatal(err)
	}
	if !synced.Recorded || synced.Historical || synced.AttemptID != attempt || synced.Observation == nil || synced.Observation.Result != model.ResultPass || synced.Observation.Seq <= 0 || synced.Cost == nil {
		t.Fatalf("sync changed recorded proof semantics: %+v", synced)
	}
	assertGraphCostNonnegative(t, *synced.Cost)
	if _, ok := synced.Cost.PhaseNS[evidencecost.RootInputPreparation]; !ok {
		t.Fatalf("sync cost excludes root/input preparation: %+v", synced.Cost.PhaseNS)
	}
	if synced.Cost.Counters[evidencecost.TestExecutionRequests] != 0 || synced.Cost.Counters[evidencecost.TestProcessesCompleted] != 0 {
		t.Fatalf("sync attributed a new test execution: %+v", synced.Cost.Counters)
	}
	for _, counter := range []evidencecost.Counter{evidencecost.GraphLoadRequests, evidencecost.GraphReadRequests, evidencecost.GraphWriteRequests, evidencecost.BundleReadRequests, evidencecost.ProfileResolutions, evidencecost.PublicationCallbacks} {
		if synced.Cost.Counters[counter] == 0 {
			t.Fatalf("sync cost lacks %s: %+v", counter, synced.Cost.Counters)
		}
	}
	header := filepath.Join(planDir, ".graph", "test-evidence", "work", attempt, "header.json")
	if raw := readEvidenceCLI(t, header); strings.Contains(raw, `"cost"`) {
		t.Fatal("cost persisted in attempt receipt")
	}
	graphAfterSync, err := os.ReadFile(graphPath)
	if err != nil {
		t.Fatal(err)
	}
	if strings.Contains(string(graphAfterSync), `"cost"`) {
		t.Fatal("cost persisted in graph")
	}

	replayJSON, err := graphCostJSON(t, "graph", "sync", "--plan", "Pilot", "--node", "work", "--attempt", attempt, "--cost", "--json")
	if exitCode(err) != 0 {
		t.Fatalf("historical replay exit=%d err=%v output=%s", exitCode(err), err, replayJSON)
	}
	var replay struct {
		Recorded    bool                  `json:"recorded"`
		Historical  bool                  `json:"historical"`
		AttemptID   string                `json:"attempt_id"`
		Observation *model.Verification   `json:"observation"`
		Cost        *evidencecost.Summary `json:"cost"`
	}
	if err := json.Unmarshal([]byte(replayJSON), &replay); err != nil {
		t.Fatal(err)
	}
	if !replay.Recorded || !replay.Historical || replay.AttemptID != attempt || replay.Observation == nil || replay.Observation.Result != synced.Observation.Result || replay.Observation.Seq != synced.Observation.Seq || replay.Cost == nil {
		t.Fatalf("unexpected historical replay: %+v", replay)
	}
	if replay.Cost.Counters[evidencecost.HistoricalReplays] == 0 || replay.Cost.Counters[evidencecost.GraphWriteRequests] != 0 || replay.Cost.Counters[evidencecost.PublicationCallbacks] != 0 || replay.Cost.Counters[evidencecost.TestExecutionRequests] != 0 {
		t.Fatalf("historical replay attributed mutation or execution: %+v", replay.Cost.Counters)
	}
	graphAfterReplay, err := os.ReadFile(graphPath)
	if err != nil {
		t.Fatal(err)
	}
	if !bytes.Equal(graphAfterSync, graphAfterReplay) {
		t.Fatal("historical replay changed the recorded graph proof")
	}

	defaultRoot := cliEvidenceFixture(t, false)
	if err := os.Chdir(defaultRoot); err != nil {
		t.Fatal(err)
	}
	defaultAttempt := graphCostCaptureAttempt(t)
	defaultJSON, err := graphCostJSON(t, "graph", "sync", "--plan", "Pilot", "--node", "work", "--by", "worker", "--attempt", defaultAttempt, "--json")
	if exitCode(err) != 0 {
		t.Fatalf("fresh default sync exit=%d err=%v output=%s", exitCode(err), err, defaultJSON)
	}
	var defaultEnvelope map[string]json.RawMessage
	if err := json.Unmarshal([]byte(defaultJSON), &defaultEnvelope); err != nil {
		t.Fatal(err)
	}
	if _, ok := defaultEnvelope["cost"]; ok {
		t.Fatalf("fresh sync without --cost emitted cost: %s", defaultJSON)
	}
	var defaultResult struct {
		Recorded    bool                `json:"recorded"`
		Historical  bool                `json:"historical"`
		Observation *model.Verification `json:"observation"`
	}
	if err := json.Unmarshal([]byte(defaultJSON), &defaultResult); err != nil {
		t.Fatal(err)
	}
	if !defaultResult.Recorded || defaultResult.Historical || defaultResult.Observation == nil || defaultResult.Observation.Result != model.ResultPass {
		t.Fatalf("fresh uncosted sync did not record a valid proof: %+v", defaultResult)
	}

	errorJSON, err := graphCostJSON(t, "graph", "sync", "--plan", "Missing", "--node", "work", "--cost", "--json")
	if exitCode(err) == 0 {
		t.Fatalf("missing-plan sync unexpectedly succeeded: %s", errorJSON)
	}
	var failed struct {
		Error string                `json:"error"`
		Cost  *evidencecost.Summary `json:"cost"`
	}
	if err := json.Unmarshal([]byte(errorJSON), &failed); err != nil {
		t.Fatalf("decode started-operation error JSON: %v\n%s", err, errorJSON)
	}
	if failed.Error == "" || failed.Cost == nil {
		t.Fatalf("started-operation error lacks error or cost: %+v", failed)
	}
	if _, ok := failed.Cost.PhaseNS[evidencecost.RootInputPreparation]; !ok {
		t.Fatalf("resolution error excludes preparation phase: %+v", failed.Cost.PhaseNS)
	}
}

func graphCostCaptureAttempt(t *testing.T) string {
	t.Helper()
	err, out := executeEvidenceCLI(t, "test", "run", "--plan", "Pilot", "--node", "work", "--by", "worker", "--phase", "green")
	if exitCode(err) != 0 {
		t.Fatalf("base test run exit=%d err=%v output=%s", exitCode(err), err, out)
	}
	fields := strings.Fields(out)
	if len(fields) < 2 {
		t.Fatalf("base test run output=%q", out)
	}
	return strings.TrimSuffix(fields[1], ":")
}

func graphCostJSON(t *testing.T, args ...string) (string, error) {
	t.Helper()
	path := filepath.Join(t.TempDir(), "stdout.json")
	f, openErr := os.OpenFile(path, os.O_CREATE|os.O_WRONLY|os.O_TRUNC, 0o600)
	if openErr != nil {
		t.Fatal(openErr)
	}
	oldStdout := os.Stdout
	os.Stdout = f
	defer func() { os.Stdout = oldStdout }()
	var commandOutput string
	err, out := executeEvidenceCLI(t, args...)
	commandOutput = out
	if closeErr := f.Close(); closeErr != nil {
		t.Fatal(closeErr)
	}
	os.Stdout = oldStdout
	stdout, readErr := os.ReadFile(path)
	if readErr != nil {
		t.Fatal(readErr)
	}
	return string(stdout) + commandOutput, err
}

func assertGraphCostNonnegative(t *testing.T, cost evidencecost.Summary) {
	t.Helper()
	if cost.TotalNS < 0 {
		t.Fatalf("negative total_ns: %d", cost.TotalNS)
	}
	for phase, nanos := range cost.PhaseNS {
		if nanos < 0 {
			t.Fatalf("negative phase %s: %d", phase, nanos)
		}
	}
}

func assertGraphCostNumericOnly(t *testing.T, raw json.RawMessage, forbidden ...string) {
	t.Helper()
	var value any
	if err := json.Unmarshal(raw, &value); err != nil {
		t.Fatal(err)
	}
	object, ok := value.(map[string]any)
	if !ok {
		t.Fatalf("cost is %T, want object", value)
	}
	for _, key := range []string{"total_ns", "phase_ns", "counters"} {
		if _, ok := object[key]; !ok {
			t.Errorf("cost object lacks %q: %s", key, raw)
		}
	}
	if len(object) != 3 {
		t.Errorf("cost object keys = %v, want only total_ns, phase_ns, counters", object)
	}
	var walk func(any)
	walk = func(v any) {
		switch typed := v.(type) {
		case map[string]any:
			for _, child := range typed {
				walk(child)
			}
		case float64:
		case string:
			t.Errorf("cost contains string value %q", typed)
		default:
			t.Errorf("cost contains nonnumeric value of type %T", typed)
		}
	}
	walk(value)
	text := string(raw)
	for _, secret := range forbidden {
		if secret != "" && strings.Contains(text, secret) {
			t.Errorf("cost contains invocation detail %q: %s", secret, text)
		}
	}
}
